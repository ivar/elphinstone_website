{{- /* Overhead photo of the property with a marker for every project that has a `location` (x, y as % from the left/top of the photo, optional label: "left") in its front matter. An optional `area` (a list of [x, y] corners, also in %) highlights the project's patch of ground and makes it clickable too. Expects the projects list page. Photo: assets/images/property-overhead.* if present. Otherwise the regional district's aerial photo of the area set by params.propertyMap in hugo.toml is fetched at build time, and the property boundary (data/property_boundary.json) is drawn over it. */ -}} {{ $image := resources.GetMatch "images/property-overhead.*" }} {{ $credit := "" }} {{ $w := 1600 }} {{ $h := 1600 }} {{ $outline := "" }} {{ with $image }} {{ $w = .Width }}{{ $h = .Height }} {{ else }} {{ $view := site.Params.propertyMap }} {{ $west := float $view.west }}{{ $east := float $view.east }} {{ $south := float $view.south }}{{ $north := float $view.north }} {{/* Web Mercator extent, so the image isn't stretched. */}} {{ $rad := div math.Pi 180 }} {{ $r := 6378137.0 }} {{ $xW := mul $r (mul $west $rad) }}{{ $xE := mul $r (mul $east $rad) }} {{ $yS := mul $r (math.Log (math.Tan (add (div math.Pi 4) (div (mul $south $rad) 2)))) }} {{ $yN := mul $r (math.Log (math.Tan (add (div math.Pi 4) (div (mul $north $rad) 2)))) }} {{ $h = int (math.Round (div (mul $w (sub $yN $yS)) (sub $xE $xW))) }} {{ $url := printf "https://maps.scrd.ca/arcgis/rest/services/Imagery_2024/ImageServer/exportImage?bbox=%.2f,%.2f,%.2f,%.2f&bboxSR=3857&imageSR=3857&size=%d,%d&format=jpg&f=image" $xW $yS $xE $yN $w $h }} {{ with try (resources.GetRemote $url) }} {{ with .Err }} {{ warnf "Could not fetch the overhead image: %s" . }} {{ else with .Value }} {{ $image = . | resources.Copy "images/property-overhead-scrd.jpg" }} {{ $credit = "Aerial photo: Sunshine Coast Regional District, 2024." }} {{ end }} {{ end }} {{/* Boundary outline in image pixels. */}} {{ range $i, $p := index site.Data.property_boundary.geometry.coordinates 0 }} {{ $x := mul $w (div (sub (index $p 0) $west) (sub $east $west)) }} {{ $y := mul $h (div (sub $north (index $p 1)) (sub $north $south)) }} {{ $outline = printf "%s%s%.1f %.1f" $outline (cond (eq $i 0) "M" "L") $x $y }} {{ end }} {{ $outline = printf "%sZ" $outline }} {{ end }}
